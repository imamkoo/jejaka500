<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # --- Kunci akses langsung ke helper deploy di dalam api/ ---
    # Terukur 2026-11-20: `/api/public/robots.txt` dilayani sebagai BERKAS
    # (ada `last-modified`, 24 byte), jadi `^api(/.*)?$` di bawah TIDAK
    # menangkap `/api/public/...`. Akibatnya setiap `.php` di `api/public/`
    # adalah endpoint publik — termasuk helper deploy lama. Itu alasannya
    # handler asli dipindah ke `api/storage/` (di luar docroot).
    #
    # Satu-satunya berkas yang boleh dieksekusi di sana adalah shim yang
    # menandatangani request, dan shim itu sendiri hanya memuat handler.
    RewriteRule ^api/public/cpanel-deploy\.php$ - [L]
    RewriteRule ^api/(public/)?deploy- [F,L]

    # --- Tolak berkas rahasia di dalam api/ ---
    RewriteRule ^api/(\.env(\..*)?|\.git/.*|composer\.(json|lock)|artisan)$ [F,L]

    # --- Rute Laravel ---
    # Catatan terukur 2026-11-20: `/css/...`, `/js/...`, `/up`, `/admin/login`
    # sudah berfungsi apa adanya (diverifikasi `content-type` CSS/JS asli, bukan
    # fallback SPA). Aturan tambahan untuk aset top-level TIDAK diperlukan dan
    # tidak ditambahkan — mengubah routing yang sudah bekerja berisiko mematikan
    # situs tanpa manfaat.
    #
    # `^api(/.*)?$` juga menangkap `/api/public/...`, sehingga berkas apa pun di
    # `api/public/` yang diminta lewat `/api/public/<nama>` dilayani Laravel dan
    # bukan sebagai berkas fisik — kecuali ditangani `[F,L]` di atas.
    RewriteRule ^api(/.*)?$      api/public/index.php [L,QSA]
    RewriteRule ^admin(/.*)?$    api/public/index.php [L,QSA]
    RewriteRule ^livewire(/.*)?$ api/public/index.php [L,QSA]
    RewriteRule ^up$             api/public/index.php [L,QSA]

    # --- Berkas/folder yang benar-benar ada dilayani langsung ---
    RewriteCond %{REQUEST_FILENAME} -f [OR]
    RewriteCond %{REQUEST_FILENAME} -d
    RewriteRule ^ - [L]

    # --- Aset SPA yang tidak ada tetap 404 ---
    RewriteCond %{REQUEST_URI} \.[a-zA-Z0-9]{1,8}$
    RewriteRule ^ - [L,R=404]

    # --- Fallback SPA ---
    RewriteRule ^ index.html [L]
</IfModule>
